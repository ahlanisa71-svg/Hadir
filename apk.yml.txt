name: Build APK
on:
  workflow_dispatch:
jobs:
  apk:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 17
      - run: bash get-face-models.sh
      - name: Siapkan proyek Capacitor
        run: |
          mkdir -p www
          cp index.html sw.js manifest.webmanifest app.css xlsx-lite.js core.js sync.js pages1.js pages2.js pages3.js app.js icon-192.png icon-512.png icon-maskable-512.png www/
          cp -r vendor www/
          npm init -y >/dev/null
          npm i @capacitor/core@6 @capacitor/cli@6 @capacitor/android@6
          echo '{"appId":"id.hadir.app","appName":"HADIR","webDir":"www","server":{"androidScheme":"https"}}' > capacitor.config.json
          npx cap add android
          sed -i 's#<application#<uses-permission android:name="android.permission.CAMERA"/>\n    <uses-feature android:name="android.hardware.camera" android:required="false"/>\n    <application#' android/app/src/main/AndroidManifest.xml
          npx cap sync android
      - name: Build
        run: cd android && ./gradlew assembleDebug --no-daemon
      - uses: actions/upload-artifact@v4
        with:
          name: hadir-apk
          path: android/app/build/outputs/apk/debug/app-debug.apk
